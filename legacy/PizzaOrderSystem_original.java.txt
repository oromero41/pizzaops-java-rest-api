import java.util.Arrays;    // Importing the Arrays class for array manipulation
import java.util.Scanner;   // Importing the Scanner class for user input

// Main class representing the Pizza Order System
public class PizzaOrderSystem {

    // Variables to keep track of order numbers, counts, and statistics
    private static int orderNum = 1;
    private static int numberOfOrders = 0;
    private static int[] sizeCount = new int[99]; // Small, Medium, Large
    private static int[] baseCount = new int[99]; // Original, Thin, Thick, Stuffed
    private static int[] sauceCount = new int[99]; // Tomato, Barbeque
    private static int[] toppingCount = new int[99]; // Toppings
    private static int[] cheeseCount = new int[99]; // Mozz, Lowfat, Vegan
    private static int[] meatCount = new int[99]; // pepperoni, ham, beef, chicken, sausage, bacon
    private static int[] vegCount = new int[99]; // pepper, olives, onions, tomato, sweetcorn, pineapple, mushroom
    private static int totalToppings = 0;
    
    // Method to update the count of pizza sizes
    private static void updateSizeCount(int[] sizeCount, int index) {
        sizeCount[index]++;
    }

    // Method to update the count of pizza bases
    private static void updateBaseCount(int[] baseCount, int index) {
        baseCount[index]++;
    }

    // Method to update the count of pizza sauces
    private static void updateSauceCount(int[] sauceCount, int index) {
        sauceCount[index]++;
    }

    // Method to update the count of cheese types
    private static void updateCheeseCount(int[] cheeseCount, int index) {
        cheeseCount[index]++;
    }

    // Main method representing the entry point of the program
public static void main(String[] args) {
    // Scanner object to take user input
    Scanner scanner = new Scanner(System.in);

// Main loop for the pizza order system
while (true) {
    // Display menu options to the user
    System.out.println("----"); // Adds a line with dashes
    System.out.println("Pizza Corner Menu:");
    System.out.println("1. Order a pizza");
    System.out.println("2. Provide statistics");
    System.out.println("3. Exit");
    System.out.println("----"); // Adds a line with dashes
    System.out.print("Enter your choice: ");

    // Validate input to ensure it's an integer
    while (!scanner.hasNextInt()) {
        System.out.println("Invalid input. Please enter a number (1, 2, or 3).");
        scanner.next(); // Consume the invalid input
    }

    int choice = scanner.nextInt();

    // Validate that the choice is 1, 2, or 3
    if (choice < 1 || choice > 3) {
        System.out.println("Invalid choice. Please enter 1, 2, or 3.");
        continue; // Restart the loop if the choice is invalid
    }

    // Switch statement to handle user choices
    switch (choice) {
        // Case 1: Order a pizza
        case 1:
            boolean addAnotherOrder = true; // Introduce a boolean variable
            while (addAnotherOrder) {
                // Display order number and process the order
                System.out.println("----"); // Adds a line with dashes
                System.out.println("Ordering pizza #" + (orderNum + 1)); // Display the correct order number
                processOrder(scanner); // Call the processOrder method

                numberOfOrders++; // Increment the number of orders

                System.out.println("Do you want to add another order? (Y/N)");
                String addAnotherOrderResponse = scanner.next().toUpperCase();
                if (addAnotherOrderResponse.equals("N")) {
                    orderNum++; // Increment orderNum only when a new pizza is ordered
                    addAnotherOrder = false; // Exit the inner loop
                    break; // Add this line to break out of the outer loop as well
                }
            }
            break;

        // Case 2: Display statistics
        case 2:
            displayStatistics();
            break;

        // Case 3: Exit the program
        case 3:
            boolean confirmExit = false;
            while (true) {
                System.out.println("Are you sure you want to exit? (Y/N)");
                String exitConfirmation = scanner.next().toUpperCase();

                if (exitConfirmation.equals("Y")) {
                    confirmExit = true;
                    break;
                } else if (exitConfirmation.equals("N")) {
                    break; // Exit the inner loop
                } else {
                    System.out.println("Invalid choice. Please enter Y or N.");
                }
            }

            if (confirmExit) {
                System.out.println("Thank you for your visit! Enjoy!");
                scanner.close();
                return; // Exit the outer loop
            }
            break;

        default:
            System.out.println("Invalid choice. Please enter 1, 2, or 3.");
            break;
    }
}



}

    // Method to process a pizza order
    private static void processOrder(Scanner scanner) {
    int currentOrder = orderNum;

    String size = getValidSize(scanner);
    updateSizeCount(sizeCount, sizeToIndex(size));

    String base = getValidBase(scanner);
    updateBaseCount(baseCount, baseToIndex(base));

    String sauce = getValidSauce(scanner);
    updateSauceCount(sauceCount, sauceToIndex(sauce));

    int[][] allToppings = getValidToppings(scanner);

    String cheese = getValidCheese(scanner);
    updateCheeseCount(cheeseCount, cheeseToIndex(cheese));

   
    // Increment the order number before displaying details
    orderNum++;

    // Display order details using the current incremented order number
    displayOrderDetails(currentOrder, size, base, sauce, allToppings, cheese, scanner);
}








// Get valid pizza size selection
private static String getValidSize(Scanner scanner) {
    String[] sizeOptions = {"S", "M", "L"}; // Fixed array for size options
    System.out.println("----"); // Adds a line with dashes
    System.out.println("Available Sizes:");
    System.out.println("S. Small");
    System.out.println("M. Medium");
    System.out.println("L. Large");
    
    while (true) {
        System.out.print("Enter pizza size (S/M/L): ");
        String sizeInput = scanner.next().toUpperCase();

        boolean isValidSize = false;
        for (String size : sizeOptions) {
            if (sizeInput.equals(size)) {
                isValidSize = true;
                break;
            }
        }

        if (isValidSize) {
            return sizeInput;
        } else {
            System.out.println("Invalid size. Please enter S, M, or L.");
        }
    }
}

// Get valid toppings information including meat and veg toppings
private static int[][] getValidToppings(Scanner scanner) {
    int[][] toppingsInfo = new int[2][]; // 2D array for toppings: meat and veg
    int[] meatToppings = new int[99];
    int[] vegToppings = new int[99];
    int additionalToppingCharge = 0; // Keep track of additional topping charge

    // Initialize the arrays with a default value of -1
    Arrays.fill(meatToppings, -1);
    Arrays.fill(vegToppings, -1);

    System.out.println("----"); // Adds a line with dashes
    System.out.println("Available Topping Categories:");
    System.out.println("1. Meat");
    System.out.println("2. Veg");

    int totalMeatToppings = 0;
    int totalVegToppings = 0;

    while (true) {
        System.out.print("Select a topping category (1-2): ");
        if (scanner.hasNextInt()) {
            int categoryChoice = scanner.nextInt();

            if (categoryChoice == 1) {
                totalMeatToppings = getValidMeatTopping(meatToppings, totalMeatToppings, scanner, totalMeatToppings + totalVegToppings);
            } else if (categoryChoice == 2) {
                totalVegToppings = getValidVegTopping(vegToppings, totalVegToppings, scanner);
            } else {
                System.out.println("Invalid input. Please enter 1 or 2.");
                continue; // Restart the loop if invalid input
            }

            // Increment additional topping charge for each topping selected beyond the initial four
            additionalToppingCharge = Math.max(0, totalMeatToppings + totalVegToppings - 4); // Charge £1 for each additional topping
            
            System.out.println("You have selected " + (totalMeatToppings + totalVegToppings) +
                    " Toppings, 4 are included in any size pizza.");
            
            System.out.println("Additional charge: £" + additionalToppingCharge);

            // Ask if the user wants to add more toppings after reaching the minimum
            if (totalMeatToppings + totalVegToppings >= 4) {
                System.out.println("----"); // Adds a line with dashes
                System.out.println("Do you want to add more toppings? (Y/N)");
                String addMoreToppings = scanner.next().toUpperCase();
                if (addMoreToppings.equals("Y")) {
                    // Continue the loop for more toppings
                } else if (addMoreToppings.equals("N")) {
                    // Exit the loop if the user doesn't want more toppings
                    break;
                } else {
                    System.out.println("Invalid choice. Please enter Y or N.");
                }
            }
        } else {
            System.out.println("Invalid input. Please enter a number.");
            scanner.next(); // Clear the invalid input
        }
    }

    toppingsInfo[0] = Arrays.copyOfRange(meatToppings, 0, totalMeatToppings);
    toppingsInfo[1] = Arrays.copyOfRange(vegToppings, 0, totalVegToppings);


    int[] chargeInfo = {additionalToppingCharge, totalMeatToppings + totalVegToppings}; // Updated to track totalToppings
    int[][] combinedInfo = {chargeInfo, toppingsInfo[0], toppingsInfo[1]};
    return combinedInfo;
}



// Get a valid selection for a meat topping from the user input
private static int getValidMeatTopping(int[] meatToppings, int meatCount, Scanner scanner, int totalToppings) {
    System.out.println("----");
    System.out.println("Select meat topping:");
    System.out.println("1. Pepperoni");
    System.out.println("2. Ham");
    System.out.println("3. Beef");
    System.out.println("4. Chicken");
    System.out.println("5. Sausage");
    System.out.println("6. Bacon");

    int toppingChoice;

    while (true) {
        System.out.print("Enter your choice: ");
        if (scanner.hasNextInt()) {
            toppingChoice = scanner.nextInt();

            // Validate the meat topping choice
            if (toppingChoice >= 1 && toppingChoice <= 6) {
                meatToppings[meatCount++] = toppingChoice; // Update the array with the topping choice
                System.out.println("Selected Meat Topping: " + getToppingName(toppingChoice));
                if (totalToppings >= 4) {
                    System.out.println("Additional topping charge: £" + (totalToppings - 4));
                }
                break; // Break the loop after a valid choice
            } else {
                System.out.println("Invalid choice. Please enter a number between 1 and 6.");
            }
        } else {
            System.out.println("Invalid input. Please enter a number.");
            scanner.next(); // Clear the invalid input
        }
    }

    return meatCount;
}

// Helper method to display toppings
private static void displayToppings(String category, int[] toppings) {
    System.out.print(category + ": [");
    boolean hasToppings = false;
    for (int j = 0; j < toppings.length; j++) {
        if (toppings[j] != 0) { // Check if the topping is selected
            String toppingName = getToppingName(toppings[j]);
            System.out.print(toppingName + ", ");
            hasToppings = true;
        }
    }
    if (hasToppings) {
        System.out.print("\b\b");
    }
    System.out.println("]");
}

// Update updateToppingCount method
private static void updateToppingCount(int[][] toppings) {
    for (int i = 0; i < toppings.length; i++) {
        for (int j = 0; j < toppings[i].length; j++) {
            if (toppings[i][j] != 0) {
                int toppingIndex = toppings[i][j] - 1;
                toppingCount[toppingIndex]++;

                // Update meat and veg counts
                if (i == 1) { // Meat toppings category
                    meatCount[toppingIndex]++;
                } else if (i == 2) { // Veg toppings category
                    vegCount[toppingIndex]++;
                }
            }
        }
    }
}

// Method to convert pizza size to its corresponding index
private static int sizeToIndex(String size) {
    switch (size) {
        case "S":
            return 0; // Small
        case "M":
            return 1; // Medium
        case "L":
            return 2; // Large
        default:
            return -1; // Invalid size
    }
}

// Method to convert pizza base to its corresponding index
private static int baseToIndex(String base) {
    switch (base) {
        case "Original":
            return 0;
        case "Thin Crust":
            return 1;
        case "Thick Crust": 
            return 2;
        case "Stuffed Crust (with Cheese)":
            return 3;
        default:
            System.out.println("Invalid base: " + base);
            return -1; // Return -1 for an invalid base
    }
}

// Method to convert pizza sauce to its corresponding index
private static int sauceToIndex(String sauce) {
    switch (sauce) {
        case "Tomato":
            return 0;
        case "Barbeque":
            return 1;
        default:
            System.out.println("Invalid sauce: " + sauce);
            return -1;
    }
}

// Method to convert the type of cheese to its corresponding index
private static int cheeseToIndex(String cheese) {
    switch (cheese) {
        case "Mozzarella (Mozz)":

            return 1;
        case "Vegan":
            return 2;
        default:
            System.out.println("Invalid cheese: " + cheese);
            return -1; // Return -1 for an invalid cheese
    }
}
